'use strict';
const fs = require('node:fs/promises'), path = require('node:path'), { createHash } = require('node:crypto');
const root = path.resolve(__dirname, '..'), sha = b => createHash('sha256').update(b).digest('hex');
const sources = ['tests/fixtures/computer-app/Program.cs', 'tests/fixtures/computer-app/TraceComputerFixture.csproj', 'scripts/smoke-computer-live.cjs'];
async function hashes() { return Object.fromEntries(await Promise.all(sources.map(async name => [name, sha(await fs.readFile(path.join(root, name)))]))); }
(async () => {
  const before = await hashes(), executable = await require('../scripts/smoke-computer-live.cjs').buildFixture();
  if (JSON.stringify(before) !== JSON.stringify(await hashes())) throw new Error('fixture_source_changed');
  const files = {};
  for (const name of ['TraceComputerFixture.exe', 'TraceComputerFixture.dll', 'TraceComputerFixture.runtimeconfig.json', 'TraceComputerFixture.deps.json'])
    files[name] = sha(await fs.readFile(path.join(path.dirname(executable), name)));
  const manifest = { schemaVersion: 1, preparedAt: new Date().toISOString(), sources: before, files, uiCreated: 0, inputActions: 0 };
  const bytes = JSON.stringify(manifest, null, 2) + '\n';
  await fs.writeFile(path.join(root, '.runtime/computer-fixture/prepared-manifest.json'), bytes);
  console.log(JSON.stringify({ manifestSha256: sha(bytes), sources: before, files, uiCreated: 0 }));
})().catch(error => { console.error(error.code || error.message); process.exitCode = 1; });
